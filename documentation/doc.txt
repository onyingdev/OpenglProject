folder structure:

 /assets            #images, models, sound, etc... this is where to place everything to program games
    /shaders
    /textures
    /scripts

 /app
    /bin            #release folder, where the actual application is

 /build             #local build objects folder

 /documentation     #documentation stuff

 /include
    /Core           #helpers, application, input manager, window. Using only the core functions the app should work
    /Render       #opengl, vertex buffer, shader loader. Manages all the graphical API config and render functions.
    /UI             #UI stuff
    /Components     #basic components for building GameObjects

 /src
    /Core
    /Render
    /UI
    main.cpp

 /vendor            #third party software
    /glad
    /GLFW
    /glm
