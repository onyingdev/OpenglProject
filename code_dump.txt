//creates triangle on window position with int size, size cant be negative
float* createTriangle(float x, float y, float length, float height, float triangle[]) {
    if (height < 0 or length < 0) {
        std::cout << "Incorrect triangle size" << std::endl;
        return triangle;
    }
    if (sizeof(*triangle) != 9) {
        std::cout << "Incorrect array size" << std::endl;
        return triangle;
    }
    triangle[0] = {x};
    triangle[1] = {y};
    triangle[2] = {0};
    triangle[3] = {x+(length/2)};
    triangle[4] = {y+height};
    triangle[5] = {0};
    triangle[6] = {x+length};
    triangle[7] = {y};
    triangle[8] = {0};
    return triangle;
}
